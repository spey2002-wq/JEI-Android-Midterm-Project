------------------------------------------------------------------------------
 1. DATABASE OVERVIEW
------------------------------------------------------------------------------

 Database Name : campus_booking_database
 Engine        : SQLite (accessed through Android Room)
 Version       : 1
 Tables        : 3  (users, study_rooms, reservations)

 Purpose:
   Lets JEI University students book study rooms on campus. A user can make
   many reservations, and a study room can be reserved many times (at
   different times). "reservations" is the junction table that links
   users <-> study_rooms (a many-to-many relationship expressed as two
   one-to-many relationships).

 Entity Relationship (text diagram):

     +-----------+ 1        N +------------------+ N        1 +--------------+
     |   users   |------------|   reservations   |------------| study_rooms  |
     +-----------+            +------------------+            +--------------+
     | user_id PK|            | reservation_id PK|            | room_id PK   |
     | ...       |            | user_id FK       |            | ...          |
     +-----------+            | room_id FK       |            +--------------+
                              +------------------+

 Legend for constraints used below:
   PK       = Primary Key
   FK       = Foreign Key
   AUTO     = AUTOINCREMENT (value generated automatically)
   NOT NULL = value is required
   UNIQUE   = no two rows may share this value

 SQLite data types used: INTEGER, TEXT
   - Dates are stored as TEXT in ISO format  : "yyyy-MM-dd"   (2026-10-28)
   - Times are stored as TEXT in 24h format  : "HH:mm"         (14:00)
   - Timestamps are stored as TEXT           : "yyyy-MM-dd HH:mm:ss"
   - Booleans are stored as INTEGER          : 0 = false, 1 = true

 ============================================================
 TABLE 3: reservations
 ============================================================
 Stores each booking made by a user for a specific study room and time slot.
 This table holds two foreign keys.

 +------------------+-----------+----------------------------+------------------------------------------+
 | Column Name      | Data Type | Key Constraints            | Description                              |
 +------------------+-----------+----------------------------+------------------------------------------+
 | reservation_id   | INTEGER   | PK, AUTO, NOT NULL         | Unique ID of the reservation             |
 | user_id          | INTEGER   | FK -> users.user_id,       | The user who made the booking            |
 |                  |           | NOT NULL                   |                                          |
 | room_id          | INTEGER   | FK -> study_rooms.room_id, | The room being booked                    |
 |                  |           | NOT NULL                   |                                          |
 | reservation_date | TEXT      | NOT NULL                   | Date of use (yyyy-MM-dd)                 |
 | start_time       | TEXT      | NOT NULL                   | Start time (HH:mm)                       |
 | end_time         | TEXT      | NOT NULL                   | End time (HH:mm)                         |
 | purpose          | TEXT      | NOT NULL                   | Reason: "Group Study", "Presentation"..  |
 | status           | TEXT      | NOT NULL                   | "CONFIRMED", "CANCELLED", "COMPLETED"    |
 | created_at       | TEXT      | NOT NULL                   | When the booking was made                |
 +------------------+-----------+----------------------------+------------------------------------------+

 Foreign key rules:
   - user_id -> users(user_id)        ON DELETE CASCADE
       (deleting a user also deletes that user's reservations)
   - room_id -> study_rooms(room_id)  ON DELETE CASCADE
       (deleting a room also deletes reservations for that room)

 SQL (for reference):

   CREATE TABLE reservations (
       reservation_id   INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
       user_id          INTEGER NOT NULL,
       room_id          INTEGER NOT NULL,
       reservation_date TEXT    NOT NULL,
       start_time       TEXT    NOT NULL,
       end_time         TEXT    NOT NULL,
       purpose          TEXT    NOT NULL,
       status           TEXT    NOT NULL DEFAULT 'CONFIRMED',
       created_at       TEXT    NOT NULL,
       FOREIGN KEY (user_id) REFERENCES users(user_id)
           ON DELETE CASCADE,
       FOREIGN KEY (room_id) REFERENCES study_rooms(room_id)
           ON DELETE CASCADE
   );
   CREATE INDEX index_reservations_user_id ON reservations(user_id);
   CREATE INDEX index_reservations_room_id ON reservations(room_id);

   import androidx.room.Index

       tableName = "reservations",

           ForeignKey(
               entity = User::class,

   data class Reservation(
       @PrimaryKey(autoGenerate = true)

       val userId: Int,

       @ColumnInfo(name = "room_id")
       val roomId: Int,

       @ColumnInfo(name = "end_time")

   package com.jei.campusbooking.data.dao

   import androidx.room.Dao
   import androidx.room.Delete
   import androidx.room.Insert
   import androidx.room.OnConflictStrategy
   import androidx.room.Query
   import androidx.room.Update
   import com.jei.campusbooking.data.entity.User
   import kotlinx.coroutines.flow.Flow

   @Dao
   interface UserDao {

       // INSERT - returns the new auto-generated user_id
       // (throws SQLiteConstraintException on duplicate email/student number)
       @Insert(onConflict = OnConflictStrategy.ABORT)
       suspend fun insertUser(user: User): Long

       // UPDATE - returns number of rows updated
       @Update
       suspend fun updateUser(user: User): Int

       @Delete
       suspend fun deleteUser(user: User): Int

       // LOGIN - find a user by email (compare password hash in code)
       @Query("SELECT * FROM users WHERE email = :email LIMIT 1")
       suspend fun getUserByEmail(email: String): User?

       @Query("SELECT * FROM users WHERE user_id = :userId")
       suspend fun getUserById(userId: Int): User?

       @Query("SELECT * FROM users WHERE student_number = :studentNumber LIMIT 1")
       suspend fun getUserByStudentNumber(studentNumber: String): User?

       // Check duplicate email during sign-up (returns 1 if exists)
       @Query("SELECT COUNT(*) FROM users WHERE email = :email")
       suspend fun countByEmail(email: String): Int

       @Query("SELECT * FROM users ORDER BY full_name ASC")
       fun getAllUsers(): Flow<List<User>>

       @Query("UPDATE users SET password_hash = :newHash WHERE user_id = :userId")
       suspend fun updatePassword(userId: Int, newHash: String): Int
   }

 ---- File: data/dao/ReservationDao.kt ---------------------------------

   package com.jei.campusbooking.data.dao

   import androidx.room.Dao
   import androidx.room.Delete
   import androidx.room.Insert
   import androidx.room.OnConflictStrategy
   import androidx.room.Query
   import androidx.room.Update
   import com.jei.campusbooking.data.entity.Reservation
   import com.jei.campusbooking.data.entity.ReservationDetail
   import kotlinx.coroutines.flow.Flow

   @Dao
   interface ReservationDao {

       @Insert(onConflict = OnConflictStrategy.ABORT)
       suspend fun insertReservation(reservation: Reservation): Long

       @Update
       suspend fun updateReservation(reservation: Reservation): Int

       @Delete
       suspend fun deleteReservation(reservation: Reservation): Int

       @Query("SELECT * FROM reservations WHERE reservation_id = :id")
       suspend fun getReservationById(id: Int): Reservation?

       // "My Reservations" screen - newest date first
       @Query("""
           SELECT * FROM reservations
           WHERE user_id = :userId
           ORDER BY reservation_date DESC, start_time DESC
       """)
       fun getReservationsByUser(userId: Int): Flow<List<Reservation>>

       // Same list but joined with room & user info for display
       @Query("""
           SELECT r.reservation_id, s.room_name, s.building, u.full_name,
                  r.reservation_date, r.start_time, r.end_time,
                  r.purpose, r.status
           FROM reservations r
           INNER JOIN study_rooms s ON r.room_id = s.room_id
           INNER JOIN users u       ON r.user_id = u.user_id
           WHERE r.user_id = :userId
           ORDER BY r.reservation_date DESC, r.start_time DESC
       """)
       fun getReservationDetailsByUser(userId: Int): Flow<List<ReservationDetail>>

       // Timetable of one room on one day (for the availability screen)
       @Query("""
           SELECT * FROM reservations
           WHERE room_id = :roomId
             AND reservation_date = :date
             AND status = 'CONFIRMED'
           ORDER BY start_time ASC
       """)
       fun getReservationsByRoomAndDate(roomId: Int, date: String): Flow<List<Reservation>>

       // Conflict check BEFORE inserting. Returns the number of overlapping
       // confirmed bookings (0 = the time slot is free).
       // Two slots overlap when: newStart < existingEnd AND newEnd > existingStart
       @Query("""
           SELECT COUNT(*) FROM reservations
           WHERE room_id = :roomId
             AND reservation_date = :date
             AND status = 'CONFIRMED'
             AND :startTime < end_time
             AND :endTime   > start_time
       """)
       suspend fun countConflicts(
           roomId: Int,
           date: String,
           startTime: String,
           endTime: String
       ): Int

       // Cancel a booking (soft delete - keeps the history)
       @Query("UPDATE reservations SET status = 'CANCELLED' WHERE reservation_id = :id")
       suspend fun cancelReservation(id: Int): Int

       @Query("UPDATE reservations SET status = :status WHERE reservation_id = :id")
       suspend fun updateStatus(id: Int, status: String): Int

       // Upcoming confirmed bookings of a user (from a given date onward)
       @Query("""
           SELECT * FROM reservations
           WHERE user_id = :userId
             AND status = 'CONFIRMED'
             AND reservation_date >= :today
           ORDER BY reservation_date ASC, start_time ASC
       """)
       fun getUpcomingReservations(userId: Int, today: String): Flow<List<Reservation>>

       // Admin: all reservations
       @Query("SELECT * FROM reservations ORDER BY reservation_date DESC, start_time DESC")
       fun getAllReservations(): Flow<List<Reservation>>
   }

 ---- File: data/AppDatabase.kt ----------------------------------------

   package com.jei.campusbooking.data

   import android.content.Context
   import androidx.room.Database
   import androidx.room.Room
   import androidx.room.RoomDatabase
   import com.jei.campusbooking.data.dao.ReservationDao
   import com.jei.campusbooking.data.dao.StudyRoomDao
   import com.jei.campusbooking.data.dao.UserDao
   import com.jei.campusbooking.data.entity.Reservation
   import com.jei.campusbooking.data.entity.StudyRoom
   import com.jei.campusbooking.data.entity.User

   @Database(
       entities = [User::class, StudyRoom::class, Reservation::class],
       version = 1,
       exportSchema = true
   )
   abstract class AppDatabase : RoomDatabase() {

       abstract fun userDao(): UserDao
       abstract fun studyRoomDao(): StudyRoomDao
       abstract fun reservationDao(): ReservationDao

       companion object {
           const val DATABASE_NAME = "campus_booking_database"

           @Volatile
           private var INSTANCE: AppDatabase? = null

           fun getDatabase(context: Context): AppDatabase {
               return INSTANCE ?: synchronized(this) {
                   val instance = Room.databaseBuilder(
                       context.applicationContext,
                       AppDatabase::class.java,
                       DATABASE_NAME
                   )
                       .fallbackToDestructiveMigration() // OK for midterm; use Migrations in production
                       .build()
                   INSTANCE = instance
                   instance
               }
           }
       }
   }

 4. USAGE EXAMPLE (ViewModel / Repository side)
------------------------------------------------------------------------------

   // Get the database
   val db = AppDatabase.getDatabase(applicationContext)

   // Book a room (call inside a coroutine, e.g. viewModelScope.launch { ... })
   suspend fun bookRoom(userId: Int, roomId: Int,
                        date: String, start: String, end: String,
                        purpose: String): Boolean {

       val conflicts = db.reservationDao()
           .countConflicts(roomId, date, start, end)

       if (conflicts > 0) return false          // time slot already taken

       db.reservationDao().insertReservation(
           Reservation(
               userId = userId,
               roomId = roomId,
               reservationDate = date,
               startTime = start,
               endTime = end,
               purpose = purpose,
               createdAt = "2026-10-26 11:53:00"   // replace with real current time
           )
       )
       return true
   }
